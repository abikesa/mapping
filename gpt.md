```python
# 🌀 Fork: Good vs Evil

def moral_dialectic(signal):
    if signal.is_selfish() and not signal.is_reflective():
        return "evil"
    elif signal.is_selfless() or signal.is_transformative():
        return "good"
    else:
        return "ambiguous"

class Signal:
    def __init__(self, intent, consequence):
        self.intent = intent
        self.consequence = consequence
    
    def is_selfish(self):
        return self.intent == "self_gain" and self.consequence == "harm"

    def is_selfless(self):
        return self.intent == "altruism"

    def is_reflective(self):
        return self.intent in ["self_awareness", "conscious_choice"]

    def is_transformative(self):
        return self.consequence in ["healing", "growth"]

# Example usage
signal = Signal("altruism", "healing")
print(moral_dialectic(signal))  # Outputs: good
```
